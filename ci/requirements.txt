# Third-party tools that CI installs by name, pinned to an exact version so a
# run's result depends on this commit and not on what PyPI served that morning.
# Dependabot watches this file (the "/ci" pip entry in .github/dependabot.yml),
# so an upgrade arrives as a PR someone reads.
#
# Only the gate's own tools belong here. The project's toolchain (ruff, pytest)
# stays in pyproject.toml's dev extras with a version bound, and CI installs it
# with `uv pip install -e '.[dev]'`.
#
# uv is the installer everywhere -- CI, the devcontainer and the Dockerfile --
# and the build frontend (`uv build`, which replaced `python -m build`). The pin
# below is the one every Python environment reads: ci.yml's setup-uv step takes
# it via `version-file: ci/requirements.txt`, and post-create.sh greps it, so
# one Dependabot PR moves both at once. The Dockerfile is the exception -- it
# pins the uv container image, which Dependabot's docker ecosystem bumps on its
# own schedule, so those two versions can sit a week apart.

uv==0.12.5    # installer, resolver, venv manager and build frontend
twine==7.0.0  # sdist/wheel metadata check before publish
